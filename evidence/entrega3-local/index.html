  10     def __init__(self, code: str, message: str, status: int = 400):

  17 REQUIRED_FIELDS = {"room_id", "student_name", "starts_at", "ends_at", "participants"}
  18 TIMESTAMP = re.compile(
  19     r"^\d{4}-\d{2}-\d{2}[Tt]\d{2}:\d{2}"
  20     r"(?::(?P<seconds>\d{2})(?:\.(?P<fraction>\d+))?)?"

  25 def parse_datetime(value: object, field: str) -> datetime:
  26     """Aceita data/hora ISO com fuso explícito e precisão de minutos."""
  27     match = TIMESTAMP.fullmatch(value) if isinstance(value, str) else None
  28     if match is None:
  29         raise DomainError("invalid_datetime", f"{field} exige data/hora ISO com fuso explícito.")
  30     seconds = match.group("seconds")
  31     fraction = match.group("fraction")
  32     if (seconds is not None and seconds != "00") or (fraction and any(c != "0" for c in fraction)):
  33         raise DomainError("invalid_datetime", f"{field} deve usar minutos inteiros (segundos iguais a zero).")
  34     try:
  35         parsed = datetime.fromisoformat(value.upper().replace("Z", "+00:00"))
  36         return parsed.astimezone(timezone.utc)
  37     except (ValueError, OverflowError) as exc:
  38         raise DomainError("invalid_datetime", f"{field} contém data/hora inválida.") from exc
  39 
  40 
  41 def format_datetime(value: datetime) -> str:
  42     """Serializa UTC em formato estável, também ordenável no SQLite."""
  43     return value.astimezone(timezone.utc).isoformat(timespec="seconds").replace("+00:00", "Z")

  46 def validate_payload(payload: object, now: datetime) -> dict:
  47     """Valida dados independentes da sala e devolve valores normalizados."""
  48     if not isinstance(payload, dict) or set(payload) != REQUIRED_FIELDS:
  49         raise DomainError(
  50             "invalid_payload",
  51             "Informe exatamente room_id, student_name, starts_at, ends_at e participants.",
  52         )
  53     room_id = payload["room_id"]
  54     if type(room_id) is not int or room_id < 1:
  55         raise DomainError("invalid_room_id", "room_id deve ser um inteiro positivo.")
  56     name = payload["student_name"]
  57     if not isinstance(name, str) or not 2 <= len(name.strip()) <= 100:
  58         raise DomainError("invalid_student_name", "student_name deve ter entre 2 e 100 caracteres após remover espaços externos.")
  59     try:
  60         name.encode("utf-8")
  61     except UnicodeEncodeError as exc:
  62         raise DomainError("invalid_student_name", "student_name deve conter caracteres Unicode válidos.") from exc
  63     participants = payload["participants"]
  64     if type(participants) is not int or participants < 1:
  65         raise DomainError("invalid_participants", "participants deve ser um inteiro positivo.")
  66     start = parse_datetime(payload["starts_at"], "starts_at")
  67     end = parse_datetime(payload["ends_at"], "ends_at")
  68     duration = (end - start).total_seconds() / 60
  69     if not 30 <= duration <= 120:
  70         raise DomainError("invalid_duration", "A reserva deve durar entre 30 e 120 minutos.")
  71     if now.tzinfo is None or now.utcoffset() is None:
  72         raise ValueError("O relógio da aplicação deve fornecer datetime com fuso horário.")
  73     if start < now:
  74         raise DomainError("start_in_past", "O início da reserva não pode estar no passado.")
  75     return {
  76         "room_id": room_id,
  77         "student_name": name.strip(),
  78         "starts_at": format_datetime(start),
  79         "ends_at": format_datetime(end),
  80         "participants": participants,
  81     }

  11 class BookingRepository:
  12     """Mantém salas e reservas sem compartilhar conexões entre threads."""
  13 
  14     def __init__(self, db_path: str | Path):
  15         self.db_path = str(db_path)
  16         if self.db_path == ":memory:":
  17             raise ValueError("Use um arquivo SQLite; bancos :memory: não são suportados.")
  18         Path(self.db_path).parent.mkdir(parents=True, exist_ok=True)
  19         with self._connection() as connection:
  20             connection.executescript(
  21                 """
  22                 CREATE TABLE IF NOT EXISTS rooms (
  23                     id INTEGER PRIMARY KEY,
  24                     name TEXT NOT NULL,
  25                     capacity INTEGER NOT NULL CHECK (capacity > 0)
  26                 );
  27                 CREATE TABLE IF NOT EXISTS bookings (
  28                     id TEXT PRIMARY KEY,
  29                     room_id INTEGER NOT NULL REFERENCES rooms(id),
  30                     student_name TEXT NOT NULL,
  31                     starts_at TEXT NOT NULL,
  32                     ends_at TEXT NOT NULL,
  33                     participants INTEGER NOT NULL CHECK (participants > 0),
  34                     status TEXT NOT NULL CHECK (status IN ('active', 'cancelled'))
  35                 );
  36                 CREATE INDEX IF NOT EXISTS bookings_room_interval
  37                     ON bookings(room_id, status, starts_at, ends_at);
  38                 INSERT OR IGNORE INTO rooms (id, name, capacity) VALUES (1, 'Sala 1', 4);
  39                 INSERT OR IGNORE INTO rooms (id, name, capacity) VALUES (2, 'Sala 2', 8);
  40                 """
  41             )
  42 
  43     @contextmanager
  44     def _connection(self):
  45         connection = sqlite3.connect(self.db_path, timeout=10)
  46         connection.row_factory = sqlite3.Row
  47         connection.execute("PRAGMA foreign_keys = ON")
  48         try:
  49             with connection:
  50                 yield connection
  51         finally:
  52             connection.close()
  53 
  54     def list_rooms(self) -> list[dict]:
  55         with self._connection() as connection:
  56             return [dict(row) for row in connection.execute("SELECT id, name, capacity FROM rooms ORDER BY id")]
  57 
  58     def list_bookings(self) -> list[dict]:
  59         with self._connection() as connection:
  60             return [dict(row) for row in connection.execute("SELECT * FROM bookings ORDER BY starts_at, id")]
  61 
  62     def create_booking(self, booking: dict) -> dict:
  63         """Valida conflito e insere na mesma transação, inclusive sob concorrência."""
  64         if booking["room_id"] > 2**63 - 1:
  65             raise DomainError("room_not_found", "Sala não encontrada.", 404)
  66         with self._connection() as connection:
  67             connection.execute("BEGIN IMMEDIATE")
  68             room = connection.execute("SELECT capacity FROM rooms WHERE id = ?", (booking["room_id"],)).fetchone()
  69             if room is None:
  70                 raise DomainError("room_not_found", "Sala não encontrada.", 404)
  71             if booking["participants"] > room["capacity"]:
  72                 raise DomainError("invalid_participants", "O número de participantes excede a capacidade da sala.")
  73             conflict = connection.execute(
  74                 """SELECT 1 FROM bookings
  75                    WHERE room_id = ? AND status = 'active'
  76                      AND starts_at < ? AND ends_at > ? LIMIT 1""",
  77                 (booking["room_id"], booking["ends_at"], booking["starts_at"]),
  78             ).fetchone()
  79             if conflict is not None:
  80                 raise DomainError("booking_conflict", "A sala já está reservada nesse intervalo.", 409)
  81             result = {"id": str(uuid4()), **booking, "status": "active"}
  82             connection.execute(
  83                 """INSERT INTO bookings
  84                    (id, room_id, student_name, starts_at, ends_at, participants, status)
  85                    VALUES (:id, :room_id, :student_name, :starts_at, :ends_at, :participants, :status)""",
  86                 result,
  87             )
  88             return result
  89 
  90     def cancel_booking(self, booking_id: str) -> dict:
  91         """Cancelar novamente mantém o resultado e não remove o histórico."""
  92         with self._connection() as connection:
  93             connection.execute("BEGIN IMMEDIATE")
  94             row = connection.execute("SELECT * FROM bookings WHERE id = ?", (booking_id,)).fetchone()
  95             if row is None:
  96                 raise DomainError("booking_not_found", "Reserva não encontrada.", 404)
  97             connection.execute("UPDATE bookings SET status = 'cancelled' WHERE id = ?", (booking_id,))
  98             return {**dict(row), "status": "cancelled"}

  18 def create_server(
  19     db_path: str | Path,
  20     host: str = "127.0.0.1",
  21     port: int = 8000,
  22     now: Callable[[], datetime] | None = None,

  24     """Cria o servidor sem bloquear; chame serve_forever para executá-lo."""

  26 
  27     class Handler(BaseHTTPRequestHandler):
  28         server_version = f"ReservaLab/{__version__}"
  29 
  30         def _send_json(self, status: int, payload: dict, allow: str | None = None):
  31             body = json.dumps(payload, ensure_ascii=False, allow_nan=False).encode("utf-8")
  32             self.send_response(status)
  33             self.send_header("Content-Type", "application/json; charset=utf-8")
  34             self.send_header("Content-Length", str(len(body)))
  35             self.send_header("Cache-Control", "no-store")
  36             if allow is not None:
  37                 self.send_header("Allow", allow)
  38             self.end_headers()
  39             if self.command != "HEAD":
  40                 self.wfile.write(body)
  41 
  42         def _read_json(self):
  43             content_type = self.headers.get("Content-Type", "").split(";", 1)[0].strip().lower()
  44             if content_type != "application/json":
  45                 raise DomainError("unsupported_media_type", "Envie Content-Type: application/json.", 415)
  46             if self.headers.get("Transfer-Encoding"):
  47                 raise DomainError("invalid_json", "Envie o tamanho do corpo em Content-Length.")
  48             try:
  49                 length = int(self.headers.get("Content-Length", ""))
  50                 if length < 0:
  51                     raise ValueError
  52             except ValueError as exc:
  53                 raise DomainError("invalid_json", "Content-Length ausente ou inválido.") from exc
  54             if length > MAX_BODY_BYTES:
  55                 raise DomainError("body_too_large", "O corpo da requisição excede 16 KiB.", 413)
  56             try:
  57                 payload = self.rfile.read(length)
  58                 if len(payload) != length:
  59                     raise ValueError("Corpo incompleto")
  60                 return json.loads(payload.decode("utf-8"), parse_constant=self._reject_constant)
  61             except (UnicodeDecodeError, ValueError, RecursionError) as exc:
  62                 raise DomainError("invalid_json", "O corpo deve conter JSON válido em UTF-8.") from exc
  63 
  64         @staticmethod
  65         def _reject_constant(value):
  66             raise ValueError(f"Constante JSON inválida: {value}")
  67 
  68         def _dispatch(self):
  69             path = urlsplit(self.path).path
  70             is_booking = path.startswith("/bookings/") and len(path.split("/")) == 3 and bool(path.split("/")[-1])
  71             routes = {"/health": "GET", "/rooms": "GET", "/bookings": "GET, POST"}
  72             allowed = "DELETE" if is_booking else routes.get(path)
  73             try:
  74                 if allowed is None:
  75                     raise DomainError("not_found", "Rota não encontrada.", 404)
  76                 if self.command not in allowed.split(", "):
  77                     self._send_json(405, {"error": {"code": "method_not_allowed", "message": "Método não permitido nessa rota."}}, allow=allowed)
  78                     return
  79                 if path == "/health":
  80                     self._send_json(200, {"status": "ok"})
  81                 elif path == "/rooms":
  82                     self._send_json(200, {"rooms": service.list_rooms()})
  83                 elif path == "/bookings" and self.command == "GET":
  84                     self._send_json(200, {"bookings": service.list_bookings()})
  85                 elif path == "/bookings":
  86                     self._send_json(201, service.create_booking(self._read_json()))
  87                 else:
  88                     self._send_json(200, service.cancel_booking(path.rsplit("/", 1)[1]))
  89             except DomainError as exc:
  90                 self._send_json(exc.status, {"error": {"code": exc.code, "message": exc.message}})
  91 
  92         do_GET = _dispatch
  93         do_POST = _dispatch
  94         do_DELETE = _dispatch
  95         do_PUT = _dispatch
  96         do_PATCH = _dispatch
  97         do_HEAD = _dispatch
  98         do_OPTIONS = _dispatch
  99         do_TRACE = _dispatch
 100 
 101     server = ThreadingHTTPServer((host, port), Handler)
 102     server.daemon_threads = True
 103     return server
